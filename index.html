(linha.c[6]?.v || 'A definir') : (linha.c[4]?.v || 'A definir'); const agenciaOponente = (oponente1 === usuarioConfere) ? (agencia2 || 'A definir') : (agencia1 || 'A definir'); const linkDoOponente = (oponente1 === usuarioConfere) ? linkOponente : linkStreamer; const linkDele = (oponente1 === usuarioConfere) ? linkStreamer : linkOponente; batalhasDoUsuario.push({ data: dia, horario: hora, oponente: nomeOponente, agenciaOponente: agenciaOponente, linkDele: linkDele, linkOponente: linkDoOponente, meta: meta }); } }); if(batalhasDoUsuario.length > 0){ document.getElementById('batalhasConfirmadas').style.display = 'block'; batalhasDoUsuario.forEach(b => { lista.innerHTML += `

📅 Data: ${b.data}

⏰ Horário: ${b.horario}

👤 Seu oponente: ${b.oponente}

🏢 Agência do oponente: ${b.agenciaOponente}

🎯 Meta: ${b.meta || 'Padrão'}

🔗 Seu link: ${b.linkDele ? `Abrir` : 'A cadastrar'}

🔗 Link do oponente: ${b.linkOponente ? `Abrir` : 'A cadastrar'}

`; }); } else { document.getElementById('batalhasConfirmadas').style.display = 'none'; } }
